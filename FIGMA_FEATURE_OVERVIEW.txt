      "frames": [

          "id": "frame_1",

│ FRONTEND                                │
│ ┌─────────────────────────────────────┐ │
│ │ Figma Token Input                   │ │  NOT STORED
│ │ (password field, not visible)       │ │  ONLY IN SESSION
│ └─────────────────────────────────────┘ │
└─────────────────────────────────────────┘
              ↓ HTTPS
┌─────────────────────────────────────────┐
│ BACKEND (main.py)                       │
│ ┌─────────────────────────────────────┐ │
│ │ Verify JWT Token                    │ │  ✅ User auth
│ │ (Authorization: Bearer ...)         │ │
│ └─────────────────────────────────────┘ │
│ ┌─────────────────────────────────────┐ │
│ │ Figma Router (figma.py)             │ │
│ │ ├─ Token used immediately           │ │  ❌ NOT STORED
│ │ ├─ Token not logged                 │ │  ❌ NOT PERSISTED
│ │ └─ Token discarded after request    │ │  ❌ NOT CACHED
│ └─────────────────────────────────────┘ │
└─────────────────────────────────────────┘
              ↓ HTTPS
┌─────────────────────────────────────────┐
│ FIGMA API (external)                    │
│ Only token is sent here                 │
└─────────────────────────────────────────┘
